export const recoveryPrefix = 'tickets-recovery-v1:';
export function draftBelongsTo(key: string, owner: DraftOwner) {
  if (!owner.id || !owner.workspace_id) return false;
  const parts = key.split(':').map(s => { try { return decodeURIComponent(s); } catch { return ''; } });
  return parts[0] === 'ticket-draft' ? parts[1] === owner.id && parts[2] === owner.workspace_id :
    ['ticket-new', 'plan-draft', 'input-draft'].includes(parts[0]) && parts[1] === owner.workspace_id && parts[2] === owner.id;
}
function decode(raw: string | null): Entry | null {
  try {
    const r = JSON.parse(raw ?? 'null') as Entry | null;
    if (!r || typeof r.key !== 'string' || typeof r.writer !== 'string' || typeof r.path !== 'string' ||
      !Number.isFinite(r.updatedAt) || Math.abs(r.updatedAt) > 8640000000000000 ||
      (r.value !== null && typeof r.value !== 'string')) return null;
    if (r.format !== 2) return { ...r, format: 2, clock: { [r.writer]: 1 } };
    if (!r.clock || typeof r.clock !== 'object' || Array.isArray(r.clock) || !Object.hasOwn(r.clock, r.writer)) return null;
    if (!Object.entries(r.clock).every(([actor, n]) => /^[a-zA-Z0-9_-]+$/.test(actor) && Number.isSafeInteger(n) && n > 0)) return null;
    return r;
  } catch { return null; }
}
